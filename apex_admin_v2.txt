function U.FuzzyMatch(hay, needle)
    if needle == "" then return true, 100 end
    hay = hay:lower(); needle = needle:lower()
    if hay == needle              then return true, 200 end
    if hay:sub(1,#needle)==needle then return true, 150 end
    if hay:find(needle,1,true)    then return true, 100 end
    local hi = 1
    for i = 1, #needle do
        local f = hay:find(needle:sub(i,i), hi, true)
        if not f then return false, 0 end
        hi = f + 1
    end
    return true, 60
end

function U.HighlightMatch(text, query)
    if query == "" then return text end
    local lo = text:lower(); local ql = query:lower()
    local s, e = lo:find(ql, 1, true)
    if s then
        return text:sub(1,s-1)
            ..'<font color="rgb(172,120,255)"><b>'
            ..text:sub(s,e).."</b></font>"
            ..text:sub(e+1)
    end
    return text
end

function U.ParseArgs(raw)
    local a = {}
    for t in raw:gmatch("%S+") do table.insert(a,t) end
    return a
end

function U.FindPlayer(name)

    local nl = name:lower()
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():sub(1,#nl) == nl then return p end
    end
    return nil
end

function Registry:Add(def)
    assert(def.aliases and #def.aliases > 0, "Need alias")
    assert(def.execute,  "Need execute")
    assert(def.category, "Need category")
    def.name   = def.name or def.aliases[1]
    def._canon = def.aliases[1]:lower()
    self._cmds[def._canon] = def
    for _, a in ipairs(def.aliases) do self._alias[a:lower()] = def._canon end
    if not self._cats[def.category] then self._cats[def.category] = {} end
    table.insert(self._cats[def.category], def._canon)
end

function Registry:Get(alias)
    local c = self._alias[alias:lower()]
    return c and self._cmds[c]
end

function Registry:Search(query, limit)
    limit = limit or 20
    local results = {}
    if query == "" then
        for _, def in pairs(self._cmds) do
            if not def.hidden then table.insert(results, {def=def, score=50}) end
        end
        table.sort(results, function(a,b)
            local ao = (Registry.CategoryDefs[a.def.category] or {}).order or 99
            local bo = (Registry.CategoryDefs[b.def.category] or {}).order or 99
            return ao ~= bo and ao < bo or a.def._canon < b.def._canon
        end)
    else
        for _, def in pairs(self._cmds) do
            if not def.hidden then
                local best = 0
                for _, alias in ipairs(def.aliases) do
                    local m, s = U.FuzzyMatch(alias, query)
                    if m and s > best then best = s end
                end
                local dm, ds = U.FuzzyMatch(def.description or "", query)
                if dm and math.floor(ds*0.4) > best then best = math.floor(ds*0.4) end
                if best > 0 then table.insert(results, {def=def, score=best}) end
            end
        end
        table.sort(results, function(a,b) return a.score > b.score end)
    end
    local out = {}
    for i = 1, math.min(#results, limit) do table.insert(out, results[i].def) end
    return out
end

function Registry:All()
    local out = {}
    for _, d in pairs(self._cmds) do if not d.hidden then table.insert(out, d) end end
    return out
end

-- HISTORY
local History = { _log={}, _cursor=0, MaxSize=200 }

function History:Push(input)
    if input == "" then return end
    if self._log[#self._log] == input then return end
    table.insert(self._log, input)
    if #self._log > self.MaxSize then table.remove(self._log, 1) end
    self._cursor = #self._log + 1
end

function History:Prev()
    self._cursor = math.max(1, self._cursor-1)
    return self._log[self._cursor] or ""
end

function History:Next()
    self._cursor = math.min(#self._log+1, self._cursor+1)
    return self._log[self._cursor] or ""
end

function History:ResetCursor() self._cursor = #self._log+1 end
function History:GetAll()      return self._log            end

-- DRAG & RESIZE
local function MakeDraggable(handle, target)
    local drag, dragStart, startPos
    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1
        or inp.UserInputType==Enum.UserInputType.Touch then
            drag=true; dragStart=inp.Position; startPos=target.Position
        end
    end)
    handle.InputEnded:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1
        or inp.UserInputType==Enum.UserInputType.Touch then drag=false end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if drag and (inp.UserInputType==Enum.UserInputType.MouseMovement
        or inp.UserInputType==Enum.UserInputType.Touch) then
            local d = inp.Position - dragStart
            target.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset+d.X,
                startPos.Y.Scale, startPos.Y.Offset+d.Y)
        end
    end)
end

local function BuildTabs()
    for _, ch in ipairs(TabScroll:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    tabButtons = {}
    local list = {{ key="All", label="All", icon="*", col=T.Violet }}
    local sorted = {}
    for k, d in pairs(Registry.CategoryDefs) do
        if Registry._cats[k] and #Registry._cats[k]>0 then
            table.insert(sorted, {key=k, def=d})
        end
    end
    table.sort(sorted, function(a,b) return a.def.order < b.def.order end)
    for _, c in ipairs(sorted) do
        table.insert(list, {key=c.key, label=c.def.label, icon=c.def.icon, col=c.def.col})
    end
    for _, tab in ipairs(list) do
        local w   = math.max(52, #(tab.icon.." "..tab.label) * 7 + 14)
        local sel = tab.key == activeTab
        local btn = U.Button({bg=sel and T.Violet or T.GlassSurf,

            size=UDim2.new(0,w,0,22), text=tab.icon.." "..tab.label,
            col=sel and T.TxtPri or T.TxtSec,
            font=T.FontBold, sz=T.SzSmall, z=13})
        U.Corner(btn, T.RPill)
        U.Stroke(btn, T.Border, 1)
        U.Ripple(btn)
        btn.MouseButton1Click:Connect(function()
            activeTab = tab.key
            BuildTabs()
            BuildCommandList(SearchBox.Text)
        end)
        btn.LayoutOrder = tab.key=="All" and 0 or (tab.def and tab.def.order or 99)
        btn.Parent = TabScroll

    end

    for _, ch in ipairs(TabScroll:GetChildren()) do
        if ch:IsA("TextButton") then tot=tot+ch.Size.X.Offset+3 end
    end
    TabScroll.CanvasSize = UDim2.new(0,tot+8,0,0)

-- EXECUTOR
local function ExecCommand(rawInput)
    rawInput = rawInput:match("^%s*(.-)%s*$")
    if rawInput=="" then return end
    History:Push(rawInput)
    History:ResetCursor()

    local stripped = rawInput:match("^:(.+)") or rawInput
    local tokens   = U.ParseArgs(stripped)
    if #tokens==0 then return end

    local key  = tokens[1]:lower()
    local args = {table.unpack(tokens,2)}
    local def  = Registry:Get(key)

    if not def then
        Notif.Push("Unknown: :"..key, "error", 2.5)
        StatusLabel.Text = "Unknown: :"..key
        return
    end

    StatusLabel.Text = "Ran: :"..def.aliases[1]

    local ok, err = pcall(def.execute, args, rawInput, {
        player=LP, notif=Notif, registry=Registry, loops=ENV.ApexLoops,
    })
    if not ok then
        Notif.Push("Error: "..tostring(err), "error", 4)
        StatusLabel.Text = "Error in :"..def.aliases[1]
    end
end

-- INPUT WIRING
local function HandleInput(text)
    local s = text:match("^:?(.*)") or ""
    if text:sub(1,1)==":" and #s>0 then
        ShowDropdown(Registry:Search(s, 8))
    else
        HideDropdown()
    end
    BuildCommandList(text)
end

SearchBox.FocusLost:Connect(function(enter)
    if enter then
        local text = SearchBox.Text:match("^%s*(.-)%s*$")
        if text:sub(1,1)==":" then
            if #dropItems>0 and DropFrame.Size.Y.Offset>0 then
                ExecCommand(":"..dropItems[selectedDropIdx].aliases[1])
            else
                ExecCommand(text)
            end
        end
        SearchBox.Text=""; HideDropdown(); BuildCommandList("")
    else
        HideDropdown()
    end
end)

UserInputService.InputBegan:Connect(function(inp, gpe)
    if gpe then return end
    if inp.KeyCode==Enum.KeyCode.F2 then
        if PanelRoot.Visible or MinBar.Visible then SetVisible(not GuiVisible)
        else PanelRoot.Visible=true; SetVisible(true) end
    end
    if not SearchBox:IsFocused() and not MinInput:IsFocused() then return end
    if inp.KeyCode==Enum.KeyCode.Up then
        if #dropItems>0 and DropFrame.Size.Y.Offset>0 then UpdateDropSel(selectedDropIdx-1)
        else local p=History:Prev(); SearchBox.Text=p; MinInput.Text=p end
    elseif inp.KeyCode==Enum.KeyCode.Down then
        if #dropItems>0 and DropFrame.Size.Y.Offset>0 then UpdateDropSel(selectedDropIdx+1)
        else local n=History:Next(); SearchBox.Text=n; MinInput.Text=n end
    elseif inp.KeyCode==Enum.KeyCode.Escape then
        HideDropdown(); SearchBox:ReleaseFocus(); MinInput:ReleaseFocus()
    elseif inp.KeyCode==Enum.KeyCode.Tab then
        if #dropItems>0 then
            SearchBox.Text=":"..dropItems[selectedDropIdx].aliases[1].." "
            SearchBox.CursorPosition=#SearchBox.Text+1
            HideDropdown()
        end
    end
end)

MinInput.FocusLost:Connect(function(enter)
    if enter then
        local t=MinInput.Text:match("^%s*(.-)%s*$")
        if t~="" then ExecCommand(t:sub(1,1)==":" and t or ":"..t) end
        MinInput.Text=""
    end
end)

Registry:Add({
    aliases={"fly","flight","flyon"},
    description="Toggle fly mode", category="Movement", usage=":fly [speed]",
    args={{name="speed",type="number",optional=true,hint="50"}},
    execute=function(args)
        State.fly = not State.fly
        local c=WaitChr()
        local root=c:FindFirstChild("HumanoidRootPart")
        if not root then Notif.Push("No root.","error"); return end
        local spd=tonumber(args[1]) or 50
        if State.fly then
            local bg=Instance.new("BodyGyro")
            bg.MaxTorque=Vector3.new(1e5,1e5,1e5); bg.D=50
            bg.Name="FlyGyro"; bg.Parent=root
            local bv=Instance.new("BodyVelocity")
            bv.Velocity=Vector3.zero; bv.MaxForce=Vector3.new(1e5,1e5,1e5)
            bv.Name="FlyVel"; bv.Parent=root
            local h=c:FindFirstChildOfClass("Humanoid")
            if h then h.PlatformStand=true end
            local conn
            conn=RunService.RenderStepped:Connect(function()
                if not State.fly then
                    conn:Disconnect()
                    pcall(function() bg:Destroy(); bv:Destroy() end)
                    if h then h.PlatformStand=false end
                    return
                end
                local mv=Vector3.zero
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then mv=mv+Cam.CFrame.LookVector  end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then mv=mv-Cam.CFrame.LookVector  end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then mv=mv-Cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then mv=mv+Cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space)     then mv=mv+Vector3.new(0,1,0)  end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then mv=mv-Vector3.new(0,1,0)  end
                bv.Velocity=mv*spd; bg.CFrame=Cam.CFrame
            end)
            Notif.Push("Fly ON (spd "..spd..")","success")
        else
            Notif.Push("Fly OFF","warn")
        end
    end,
})

                        for _,p in ipairs(Players:GetPlayers()) do
                            if p~=LP then
                                local tr=U.GetRoot(p)
                                if tr and (tr.Position-r.Position).Magnitude<=range then
                                    local h=U.GetHum(p)
                                    if h then h.Health=h.Health-10 end
                                end
                            end
                        end
                    end
                    task.wait(0.2)
                end
            end)
            Notif.Push("KillAura ON (range "..range..")","success")
        else
            Notif.Push("KillAura OFF","warn")
        end
    end,
})

            Cam.CFrame=CFrame.new(Cam.CFrame.Position, tr.Position+Vector3.new(0,2,0))
            task.delay(0.05, function() Cam.CameraType=Enum.CameraType.Custom end)
            Notif.Push("Locked onto "..nearest.Name,"success")
        end
    end,
})
